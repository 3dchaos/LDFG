[@军鼓wizard_开启_电耗子]
{
#IF
#ACT
GOTO @军鼓wizard_刷新_电耗子
}

[@军鼓wizard_刷新_电耗子]
{
#IF
#ACT
#CALL [系统功能\军鼓流派\法师军鼓.txt] @军鼓wizard_品质基础

#IF
LARGE N$军鼓品质 0
#ACT
LockUpdateAbil
ChangeHumAbility 7 = <$STR(N$军鼓点)> 3
ChangeHumAbility 8 = <$STR(N$军鼓大点)> 3
ChangeHumAbility 18 = <$STR(N$军鼓大点)> 3
AddHumNewValue 0 = <$STR(N$军鼓元素)> 3
AddHumNewValue 1 = <$STR(N$军鼓元素)> 3
UpdateAbil

#IF
LARGE N$军鼓品质 3
#ACT
MOV N$军鼓电痕 1

#IF
LARGE N$军鼓品质 5
#ACT
MOV N$军鼓雷网 1
}

[@军鼓wizard_关闭_电耗子]
{
#IF
#ACT
MOV N$军鼓电痕 0
MOV N$军鼓雷网 0
}

[@军鼓wizard_攻击_电耗子]
{
#IF
CHECKMAGICNAME 雷电术
RANDOMEX <$STR(N$军鼓触发几率)> 100
#ACT
ChangeDamageValue 1 + <$STR(N$军鼓百分)>
MOV N$军鼓触发成功 1
MOV S$军鼓触发说明 攻击触发：电耗子几率强化雷电术，雷系专精已结算。

#IF
CHECKMAGICNAME 疾光电影
RANDOMEX <$STR(N$军鼓触发几率)> 100
#ACT
ChangeDamageValue 1 + <$STR(N$军鼓大点)>
MOV N$军鼓触发成功 1
MOV S$军鼓触发说明 攻击触发：电耗子几率强化疾光电影，雷线加码已结算。

#IF
CHECKMAGICNAME 雷电术
LARGE N$军鼓品质 4
CHECKCURRTARGETRACE = 0
RANDOMEX <$STR(N$军鼓触发几率)> 100
#ACT
M.MOV S$军鼓雷痕 开启
M.SetArrBuff 3 8 1 290 6 0 0 0 {电耗子|243}\雷痕短暂缠身，连续续上才会保持
MOV N$军鼓触发成功 1
MOV S$军鼓触发说明 攻击触发：电耗子短暂雷痕已挂上。

#IF
CHECKMAGICNAME 疾光电影
LARGE N$军鼓品质 4
CHECKCURRTARGETRACE = 0
RANDOMEX <$STR(N$军鼓触发几率)> 100
#ACT
M.MOV S$军鼓雷痕 开启
M.SetArrBuff 3 8 1 290 6 0 0 0 {电耗子|243}\雷痕短暂缠身，连续续上才会保持
MOV N$军鼓触发成功 1
MOV S$军鼓触发说明 攻击触发：电耗子短暂雷痕已挂上。

#IF
CHECKMAGICNAME 雷电术
LARGE N$军鼓品质 4

RANDOMEX <$STR(N$军鼓触发几率)> 100
#ACT

MOV N$军鼓触发成功 1
MOV S$军鼓触发说明 攻击触发：电耗子短暂雷痕已显形。

#IF

NOT CHECKCURRTARGETRACE = 0

#ACT

MOV N$军鼓触发成功 1
