[@军鼓wizard_开启_电耗子]
{
#IF
#ACT
GOTO @军鼓wizard_刷新_电耗子
}

[@军鼓wizard_刷新_电耗子]
{
#IF
#ACT
#CALL [系统功能\军鼓流派\法师军鼓.txt] @军鼓wizard_品质基础

#IF
LARGE N$军鼓品质 0
#ACT
LockUpdateAbil
ChangeHumAbility 7 = <$STR(N$军鼓点)> 3
ChangeHumAbility 8 = <$STR(N$军鼓大点)> 3
ChangeHumAbility 18 = <$STR(N$军鼓大点)> 3
AddHumNewValue 0 = <$STR(N$军鼓元素)> 3
AddHumNewValue 1 = <$STR(N$军鼓元素)> 3
UpdateAbil

#IF
LARGE N$军鼓品质 3
#ACT
MOV N$军鼓电痕 1

#IF
LARGE N$军鼓品质 5
#ACT
MOV N$军鼓雷网 1
}

[@军鼓wizard_关闭_电耗子]
{
#IF
#ACT
MOV N$军鼓电痕 0
MOV N$军鼓雷网 0
}

[@军鼓wizard_攻击_电耗子]
{
#IF
CHECKMAGICNAME 雷电术
#ACT
ChangeDamageValue 1 + <$STR(N$军鼓百分)>
MOV N$军鼓触发成功 1
MOV S$军鼓触发说明 攻击触发：电耗子强化雷电术，雷系专精已结算。

#IF
CHECKMAGICNAME 疾光电影
#ACT
ChangeDamageValue 1 + <$STR(N$军鼓大点)>
MOV N$军鼓触发成功 1
MOV S$军鼓触发说明 攻击触发：电耗子强化疾光电影，雷线加码已结算。

#IF
LARGE N$军鼓品质 4
#ACT
M.MOV S$军鼓雷痕 开启
MOV N$军鼓触发成功 1
MOV S$军鼓触发说明 攻击触发：电耗子给目标挂上雷痕。

#IF
LARGE N$军鼓品质 4
CHECKCURRTARGETRACE = 0
#ACT
M.SetArrBuff 3 8 1 290 10 0 0 0 {电耗子|243}\雷痕落在了身上，下一次雷系更疼

#IF
LARGE N$军鼓品质 4
NOT CHECKCURRTARGETRACE = 0
#ACT
M.SETICON 1 0 290 44 -30 1 0 0 150 0
}

MOV S$军鼓触发说明 受击触发：小冰箱冰甲减伤已结算。

LARGE N$军鼓品质 4
#ACT

LARGE N$军鼓品质 5
#ACT
