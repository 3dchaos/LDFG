{

}

[@军鼓taoist_刷新_养娃人]
{
#IF
#ACT
#CALL [系统功能\军鼓流派\道士军鼓.txt] @军鼓taoist_品质基础

#IF
LARGE N$军鼓品质 0
#ACT

ChangeHumAbility 10 = <$STR(N$军鼓大点)> 3

#ACT

#IF
LARGE N$军鼓品质 5
#ACT

}

{

MOV N$军鼓宝宝联动 0
}

[@军鼓taoist_攻击_养娃人]
{
#IF
CHECKSLAVECOUNT > 0

#ACT
ChangeDamageValue 1 + <$STR(N$军鼓百分)>
MOV N$军鼓触发成功 1
MOV S$军鼓触发说明 攻击触发：养娃人人宠联动伤害已结算。

#IF
CHECKSLAVECOUNT > 0
LARGE N$军鼓品质 4

#ACT
