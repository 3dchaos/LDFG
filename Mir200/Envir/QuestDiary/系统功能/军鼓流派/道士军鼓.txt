[@军鼓taoist_开启_养娃人]
{
#IF
#ACT
GOTO @军鼓taoist_刷新_养娃人
}

[@军鼓taoist_刷新_养娃人]
{
#IF
#ACT
#CALL [系统功能\军鼓流派\道士军鼓.txt] @军鼓taoist_品质基础

#IF
LARGE N$军鼓品质 0
#ACT
LockUpdateAbil
ChangeHumAbility 9 = <$STR(N$军鼓点)> 3
ChangeHumAbility 10 = <$STR(N$军鼓大点)> 3
AddHumNewValue 24 = <$STR(N$军鼓元素)> 3
UpdateAbil

#IF
LARGE N$军鼓品质 3
#ACT
MOV N$军鼓宝宝成长 1

#IF
LARGE N$军鼓品质 5
#ACT
MOV N$军鼓宝宝联动 1
}

[@军鼓taoist_关闭_养娃人]
{
#IF
#ACT
MOV N$军鼓宝宝成长 0
MOV N$军鼓宝宝联动 0
}

[@军鼓taoist_攻击_养娃人]
{
#IF
CHECKSLAVECOUNT > 0
#ACT
ChangeDamageValue 1 + <$STR(N$军鼓百分)>

#IF
CHECKSLAVECOUNT > 0

#ACT
ChangeDamageValue 1 + <$STR(N$军鼓顶级加码)>
}

{

#IF

#ACT

#IF

#ACT
