[@军鼓warrior_攻击_拆门板]
{
#IF
NOT CHECKCURRTARGETRACE = 0
RANDOMEX <$STR(N$军鼓触发几率)> 100
#ACT
M.HumanHP - <$STR(N$军鼓切割)> 0 1 0 0 3
MOV N$军鼓触发成功 1
MOV S$军鼓触发说明 攻击触发：拆门板断岳切割，对怪物造成少量最大生命伤害。
BREAK
}

[@军鼓warrior_攻击_剁馅机]

[@军鼓warrior_受击_铁头娃]
{
#IF
RANDOMEX <$STR(N$军鼓触发几率)> 100
#ACT
ChangeDamageValue 1 - <$STR(N$军鼓铁头减伤)>
MOV N$军鼓触发成功 1
MOV S$军鼓触发说明 受击触发：铁头娃山岳护体，本次伤害已降低。

#IF
EQUAL N$军鼓触发成功 1
LARGE N$军鼓品质 4
#ACT
HumanHP + <$STR(N$军鼓铁头回血)> 0 1 0 0 1
MOV S$军鼓触发说明 受击触发：铁头娃山岳护体，本次伤害降低并小幅回血。
BREAK
}
