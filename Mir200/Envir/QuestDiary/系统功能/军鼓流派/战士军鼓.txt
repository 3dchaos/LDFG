[@军鼓warrior_开启_拆门板]
{
#IF
#ACT
GOTO @军鼓warrior_刷新_拆门板
}

[@军鼓warrior_刷新_拆门板]
{
#IF
#ACT
#CALL [系统功能\军鼓流派\战士军鼓.txt] @军鼓warrior_品质基础

#IF
LARGE N$军鼓品质 0
#ACT
LockUpdateAbil
ChangeHumAbility 5 = <$STR(N$军鼓点)> 3
ChangeHumAbility 6 = <$STR(N$军鼓大点)> 3
AddHumNewValue 0 = <$STR(N$军鼓元素)> 3
AddHumNewValue 4 = <$STR(N$军鼓元素)> 3
UpdateAbil

#IF
LARGE N$军鼓品质 2
#ACT
MOV N$军鼓拆门板 1

#IF
LARGE N$军鼓品质 4
#ACT
MOV N$军鼓拆门板连锁 1
}

[@军鼓warrior_关闭_拆门板]
{
#IF
#ACT
MOV N$军鼓拆门板 0
MOV N$军鼓拆门板连锁 0
}

[@军鼓warrior_攻击_拆门板]
{
#IF
CHECKMAGICNAME 烈火剑法
RANDOMEX <$STR(N$军鼓触发几率)> 100
#ACT
ChangeDamageValue 1 + <$STR(N$军鼓百分)>
MOV N$军鼓触发成功 1
MOV S$军鼓触发说明 攻击触发：拆门板强化烈火剑法，爆发破防已结算。

#IF
CHECKMAGICNAME 开天斩
RANDOMEX <$STR(N$军鼓触发几率)> 100
#ACT
ChangeDamageValue 1 + <$STR(N$军鼓大点)>
MOV N$军鼓触发成功 1
MOV S$军鼓触发说明 攻击触发：拆门板强化开天斩，大点破防已结算。

#IF
CHECKMAGICNAME 逐日剑法
RANDOMEX <$STR(N$军鼓触发几率)> 100
#ACT
ChangeDamageValue 1 + <$STR(N$军鼓元素)>
MOV N$军鼓触发成功 1
MOV S$军鼓触发说明 攻击触发：拆门板强化逐日剑法，元素加码已结算。

#IF
LARGE N$军鼓品质 5
RANDOMEX <$STR(N$军鼓触发几率)> 100
#ACT
ChangeDamageValue 1 + <$STR(N$军鼓顶级加码)>
MOV N$军鼓触发成功 1
MOV S$军鼓触发说明 攻击触发：拆门板封顶加码已结算。
}
